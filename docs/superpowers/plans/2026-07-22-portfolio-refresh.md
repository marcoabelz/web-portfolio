# Portfolio Refresh Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the outdated education-first portfolio with a fast, responsive, accessible, light/dark production-engineering portfolio driven by Marco's approved CV and LinkedIn content.

**Architecture:** Keep the existing Next.js 14 App Router application and refactor it incrementally into typed content modules, Server Component sections, a shared static case-study route, and a separate project archive. Browser JavaScript is limited to navigation state and persisted theme control; all portfolio content and metadata remain statically renderable.

**Tech Stack:** Next.js 14.2, React 18, TypeScript 5, Tailwind CSS 3, `next/image`, `next/font`, Lucide React, Vitest, pnpm.

const LOCAL_SITE_URL = "http://localhost:3000";

function getLocalFallback(environment: string | undefined): URL {
  if (environment === "production") {
    throw new Error("NEXT_PUBLIC_SITE_URL must be a valid HTTP(S) origin in production.");
  }

  return new URL(LOCAL_SITE_URL);
}

export function getSiteUrl(
  value: string | undefined,
  environment: string | undefined = process.env.NODE_ENV,
): URL {
  if (!value) {
    return getLocalFallback(environment);
  }

  let parsed: URL;

  try {
    parsed = new URL(value);
  } catch {
    return getLocalFallback(environment);
  }

  if (parsed.protocol !== "http:" && parsed.protocol !== "https:") {
    return getLocalFallback(environment);
  }

  return new URL(parsed.origin);
}

import { Moon, Sun } from "lucide-react";
import { useEffect, useState } from "react";

export function ThemeToggle() {
  const [dark, setDark] = useState(false);

  useEffect(() => setDark(document.documentElement.classList.contains("dark")), []);

  const toggle = () => {
    const next = !dark;
    document.documentElement.classList.toggle("dark", next);
    localStorage.setItem("portfolio-theme", next ? "dark" : "light");
    setDark(next);
  };

  return (
    <button type="button" onClick={toggle} aria-label={`Switch to ${dark ? "light" : "dark"} mode`} className="grid size-10 place-items-center rounded-full border border-line bg-surface text-ink transition-colors duration-200 hover:border-accent hover:text-accent">
      {dark ? <Sun className="size-4" aria-hidden="true" /> : <Moon className="size-4" aria-hidden="true" />}

}

import type { Metadata } from "next";
import { notFound } from "next/navigation";
import { CaseStudy } from "@/components/work/CaseStudy";
import { SELECTED_PROJECTS, getProject } from "@/data/projects";
import { SITE } from "@/lib/site";

type Props = { params: { slug: string } };

export function generateStaticParams() {
  return SELECTED_PROJECTS.map(({ slug }) => ({ slug }));
}

export function generateMetadata({ params }: Props): Metadata {
  const project = getProject(params.slug);
  if (!project) return {};
  const title = `${project.name} Case Study`;
  const description = `${project.descriptor}. ${project.outcome}`;
  const canonicalUrl = new URL(`/work/${project.slug}`, SITE.url).href;
  const sharedImageUrl = new URL("/opengraph-image", SITE.url).href;
  return {
    title,
    description,
    alternates: { canonical: canonicalUrl },
    openGraph: { title, description, url: canonicalUrl, type: "article", images: [sharedImageUrl] },
    twitter: { card: "summary_large_image", title, description, images: [sharedImageUrl] },
  };
}

export default function ProjectPage({ params }: Props) {
  const project = getProject(params.slug);
  if (!project) notFound();
  return <CaseStudy project={project} />;
}

- [ ] **Step 3: Add robots and sitemap endpoints from the canonical configuration**

Create `client/src/app/robots.ts`:

```ts
import type { MetadataRoute } from "next";
import { SITE } from "@/lib/site";

export default function robots(): MetadataRoute.Robots {
  return { rules: { userAgent: "*", allow: "/" }, sitemap: new URL("/sitemap.xml", SITE.url).href };
}
```

Create `client/src/app/sitemap.ts`:

```ts
import type { MetadataRoute } from "next";
import { SELECTED_PROJECTS } from "@/data/projects";
import { SITE } from "@/lib/site";

export default function sitemap(): MetadataRoute.Sitemap {
  const paths = ["/", "/work/archive", ...SELECTED_PROJECTS.map((project) => `/work/${project.slug}`)];
  return paths.map((path) => ({ url: new URL(path, SITE.url).href, changeFrequency: path === "/" ? "monthly" : "yearly", priority: path === "/" ? 1 : 0.7 }));
}
```
